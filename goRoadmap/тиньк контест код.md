package main

  

import (
"bufio"
"fmt"
"os"
)

  

func main() {

scanner := bufio.NewScanner(os.Stdin)
scanner.Buffer(make([]byte, 0, 1<<20+16), 1<<20+16)
scanner.Scan()
s := scanner.Text()
n := len(s)
p := 0

for p < n && s[p] == 'a' {
p++
}

if p == n {
fmt.Println("Yes")
return
}

  

q := 0
for q < n-p && s[n-1-q] == 'a' {
q++
}

  

if q < p {
fmt.Println("No")
return
}

  

l, r := p, n-1-q
for l < r {
if s[l] != s[r] {
fmt.Println("No")
return
}
l++
r--
}

fmt.Println("Yes")

}

package main

  

import (
"bufio"
"fmt"
"os"
)

const MOD int64 = 1_000_000_007

func powMod(base, exp int64) int64 {

base %= MOD

if base < 0 {

base += MOD

}

result := int64(1)

for exp > 0 {

if exp&1 == 1 {

result = result * base % MOD

}

base = base * base % MOD

exp >>= 1

}

return result

}

  

func main() {

reader := bufio.NewReader(os.Stdin)

writer := bufio.NewWriter(os.Stdout)

defer writer.Flush()

  

var n, m int64
fmt.Fscan(reader, &n, &m)

mMod := m % MOD
first := mMod * ((m - 1) % MOD) % MOD
trans := (mMod*mMod%MOD - 3*mMod%MOD + 3) % MOD
if trans < 0 {
trans += MOD
}
ans := first * powMod(trans, n-1) % MOD
fmt.Fprintln(writer, ans)
}

package main

  

import (

"bufio"

"os"

"strconv"

)

  

var reader *bufio.Reader

  

func readInt() int {

n := 0

c, _ := reader.ReadByte()

for ; c == ' ' || c == '\n' || c == '\r' || c == '\t'; c, _ = reader.ReadByte() {

}

for ; '0' <= c && c <= '9'; c, _ = reader.ReadByte() {

n = n*10 + int(c-'0')

}

return n

}

  

func main() {

reader = bufio.NewReaderSize(os.Stdin, 1<<20)

writer := bufio.NewWriterSize(os.Stdout, 1<<20)

defer writer.Flush()

  

n := readInt()

  

const negInf = int64(-1) << 62

buy1, sell1 := negInf, int64(0)

buy2, sell2 := negInf, int64(0)

  

for i := 0; i < n; i++ {

p := int64(readInt())

  

if -p > buy1 {

buy1 = -p

}

if buy1+p > sell1 {

sell1 = buy1 + p

}

if sell1-p > buy2 {

buy2 = sell1 - p

}

if buy2+p > sell2 {

sell2 = buy2 + p

}

}

  

writer.WriteString(strconv.FormatInt(sell2, 10))

writer.WriteByte('\n')

}

package main

  

import (

"bufio"

"fmt"

"os"

)

  

func main() {

reader := bufio.NewReader(os.Stdin)

var n int

fmt.Fscan(reader, &n)

var s string

fmt.Fscan(reader, &s)

  

isTOI := func(c byte) bool {

return c == 'T' || c == 'O' || c == 'I'

}

  

matchCost := func(a, b byte) int {

best := 3

for _, c := range []byte{'T', 'O', 'I'} {

cost := 0

if a != c {

cost++

}

if b != c {

cost++

}

if cost < best {

best = cost

}

}

return best

}

  

if n == 1 {

if isTOI(s[0]) {

fmt.Println(0)

} else {

fmt.Println(1)

}

return

}

  

dp := make([][]int, n)

for i := range dp {

dp[i] = make([]int, n)

}

  

for i := 0; i < n; i++ {

if isTOI(s[i]) {

dp[i][i] = 0

} else {

dp[i][i] = 1

}

}

  

for length := 2; length <= n; length++ {

for i := 0; i+length-1 < n; i++ {

j := i + length - 1

  

inner := 0

if i+1 <= j-1 {

inner = dp[i+1][j-1]

}

best := matchCost(s[i], s[j]) + inner

  

if v := 1 + dp[i+1][j]; v < best {

best = v

}

if v := 1 + dp[i][j-1]; v < best {

best = v

}

dp[i][j] = best

}

}

  

fmt.Println(dp[0][n-1])

}

package main

  

import (

"bufio"

"container/heap"

"fmt"

"os"

"sort"

"strconv"

)

  

// Min-heap для int64

type MinHeap []int64

  

func (h MinHeap) Len() int { return len(h) }

func (h MinHeap) Less(i, j int) bool { return h[i] < h[j] }

func (h MinHeap) Swap(i, j int) { h[i], h[j] = h[j], h[i] }

func (h *MinHeap) Push(x interface{}) { *h = append(*h, x.(int64)) }

func (h *MinHeap) Pop() interface{} {

old := *h

n := len(old)

x := old[n-1]

*h = old[:n-1]

return x

}

  

var scanner *bufio.Scanner

  

func readInt() int {

scanner.Scan()

v, _ := strconv.Atoi(scanner.Text())

return v

}

  

func readInt64() int64 {

scanner.Scan()

v, _ := strconv.ParseInt(scanner.Text(), 10, 64)

return v

}

  

func main() {

scanner = bufio.NewScanner(os.Stdin)

scanner.Buffer(make([]byte, 1<<20), 1<<24)

scanner.Split(bufio.ScanWords)

  

writer := bufio.NewWriter(os.Stdout)

defer writer.Flush()

  

a := readInt()

b := readInt()

n := readInt()

  

cands := make([][2]int64, n)

for i := 0; i < n; i++ {

cands[i][0] = readInt64() // x — backend skill

cands[i][1] = readInt64() // y — ML skill

}

  

// Сортировка по убыванию x - y.

// После этой сортировки в оптимальном ответе сначала идут backend-выбранные,

// затем ML-выбранные (доказывается обменным аргументом).

sort.Slice(cands, func(i, j int) bool {

return (cands[i][0] - cands[i][1]) > (cands[j][0] - cands[j][1])

})

  

// f[k] — максимальная сумма x по топ-a кандидатам из первых k (k = 0..n).

f := make([]int64, n+1)

{

h := &MinHeap{}

var sum int64

for k := 0; k < n; k++ {

x := cands[k][0]

if a > 0 {

if h.Len() < a {

heap.Push(h, x)

sum += x

} else if x > (*h)[0] {

sum += x - (*h)[0]

(*h)[0] = x

heap.Fix(h, 0)

}

}

f[k+1] = sum

}

}

  

// g[k] — максимальная сумма y по топ-b кандидатам из cands[k..n-1] (k = 0..n).

g := make([]int64, n+1)

{

h := &MinHeap{}

var sum int64

for k := n - 1; k >= 0; k-- {

y := cands[k][1]

if b > 0 {

if h.Len() < b {

heap.Push(h, y)

sum += y

} else if y > (*h)[0] {

sum += y - (*h)[0]

(*h)[0] = y

heap.Fix(h, 0)

}

}

g[k] = sum

}

// g[n] = 0 по умолчанию (нужно когда b = 0 и k = n).

}

  

var ans int64

first := true

// Точка разреза k: первые k кандидатов — кандидаты в backend, последние n-k — в ML.

// Должно выполняться k >= a и n - k >= b.

for k := a; k <= n-b; k++ {

v := f[k] + g[k]

if first || v > ans {

ans = v

first = false

}

}

  

fmt.Fprintln(writer, ans)

}
