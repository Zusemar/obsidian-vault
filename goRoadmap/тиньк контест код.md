```go
package main

  

import (
"bufio"
"fmt"
"os"
)

  

func main() {

scanner := bufio.NewScanner(os.Stdin)
scanner.Buffer(make([]byte, 0, 1<<20+16), 1<<20+16)
scanner.Scan()
s := scanner.Text()
n := len(s)
p := 0

for p < n && s[p] == 'a' {
p++
}

if p == n {
fmt.Println("Yes")
return
}

  

q := 0
for q < n-p && s[n-1-q] == 'a' {
q++
}

  

if q < p {
fmt.Println("No")
return
}

  

l, r := p, n-1-q
for l < r {
if s[l] != s[r] {
fmt.Println("No")
return
}
l++
r--
}

fmt.Println("Yes")

}
```

```go
package main

  

import (
"bufio"
"fmt"
"os"
)

const MOD int64 = 1_000_000_007

func powMod(base, exp int64) int64 {

base %= MOD

if base < 0 {

base += MOD

}

result := int64(1)

for exp > 0 {

if exp&1 == 1 {

result = result * base % MOD

}

base = base * base % MOD

exp >>= 1

}

return result

}

  

func main() {

reader := bufio.NewReader(os.Stdin)

writer := bufio.NewWriter(os.Stdout)

defer writer.Flush()

  

var n, m int64
fmt.Fscan(reader, &n, &m)

mMod := m % MOD
first := mMod * ((m - 1) % MOD) % MOD
trans := (mMod*mMod%MOD - 3*mMod%MOD + 3) % MOD
if trans < 0 {
trans += MOD
}
ans := first * powMod(trans, n-1) % MOD
fmt.Fprintln(writer, ans)
}
```

```go
package main

  

import (

"bufio"

"os"

"strconv"

)

  

var reader *bufio.Reader

  

func readInt() int {

n := 0

c, _ := reader.ReadByte()

for ; c == ' ' || c == '\n' || c == '\r' || c == '\t'; c, _ = reader.ReadByte() {

}

for ; '0' <= c && c <= '9'; c, _ = reader.ReadByte() {

n = n*10 + int(c-'0')

}

return n

}

  

func main() {

reader = bufio.NewReaderSize(os.Stdin, 1<<20)

writer := bufio.NewWriterSize(os.Stdout, 1<<20)

defer writer.Flush()

  

n := readInt()

  

const negInf = int64(-1) << 62

buy1, sell1 := negInf, int64(0)

buy2, sell2 := negInf, int64(0)

  

for i := 0; i < n; i++ {

p := int64(readInt())

  

if -p > buy1 {

buy1 = -p

}

if buy1+p > sell1 {

sell1 = buy1 + p

}

if sell1-p > buy2 {

buy2 = sell1 - p

}

if buy2+p > sell2 {

sell2 = buy2 + p

}

}

  

writer.WriteString(strconv.FormatInt(sell2, 10))

writer.WriteByte('\n')

}
```
